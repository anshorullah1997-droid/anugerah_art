create or replace function public.verify_admin_pin(p_pin_hash text)
returns boolean language plpgsql security definer set search_path = public as $$
declare v_h text;
begin
  select pin_hash into v_h from public.anugerah_admin_auth where id = 'default';
  return coalesce(v_h, '8d969eef6ecad3c29a3a629280e686cf0c3f5d5a86aff3ca12020c923adc6c92') = p_pin_hash;
end; $$;
grant execute on function public.verify_admin_pin(text) to anon, authenticated;

drop function if exists public.save_store_secure(text, text, jsonb);
create or replace function public.save_store_secure(
  p_pin_hash text, p_new_pin_hash text, p_payload jsonb,
  p_store_id text default 'main_store', p_snapshot_label text default 'Update WebApp Builder'
) returns jsonb language plpgsql security definer set search_path = public as $$
declare
  v_h text;
  v_id text := coalesce(nullif(regexp_replace(trim(p_store_id), '[^a-zA-Z0-9_-]', '_', 'g'), ''), 'main_store');
  v_clean jsonb := p_payload;
begin
  select pin_hash into v_h from public.anugerah_admin_auth where id = 'default';
  if v_h is not null and v_h != p_pin_hash then
    raise exception 'PIN Admin 6 Digit Tidak Valid!';
  end if;
  if p_new_pin_hash is not null and length(trim(p_new_pin_hash)) = 64 then
    insert into public.anugerah_admin_auth (id, pin_hash, updated_at) values ('default', trim(p_new_pin_hash), now())
    on conflict (id) do update set pin_hash = excluded.pin_hash, updated_at = now();
  end if;
  if v_clean ? 'settings' then
    v_clean := jsonb_set(v_clean, '{settings}', (v_clean -> 'settings') - 'admin_pin' - 'admin_pin_hash');
  end if;
  insert into public.anugerah_store (id, payload, pin_hash, updated_at) values (v_id, v_clean, null, now())
  on conflict (id) do update set payload = excluded.payload, pin_hash = null, updated_at = now();
  insert into public.anugerah_store_history (store_id, label, payload)
  values (v_id, coalesce(nullif(trim(p_snapshot_label), ''), 'Auto Snapshot'), v_clean);
  delete from public.anugerah_store_history where history_id in (
    select history_id from public.anugerah_store_history where store_id = v_id order by created_at desc offset 25
  );
  return jsonb_build_object('status', 'success', 'store_id', v_id, 'updated_at', now());
end; $$;
grant execute on function public.save_store_secure(text, text, jsonb, text, text) to anon, authenticated;

create or replace function public.list_store_workspaces()

language sql security definer set search_path = public as $$
  select s.id, coalesce(s.payload -> 'settings' ->> 'store_name', s.id), s.updated_at
  from public.anugerah_store s order by (s.id = 'main_store') desc, s.updated_at desc limit 30;
$$;
grant execute on function public.list_store_workspaces() to anon, authenticated;

create or replace function public.delete_store_workspace(p_pin_hash text, p_store_id text)
returns jsonb language plpgsql security definer set search_path = public as $$
declare v_h text;
begin
  select pin_hash into v_h from public.anugerah_admin_auth where id = 'default';
  if v_h is not null and v_h != p_pin_hash then raise exception 'PIN Tidak Valid!'; end if;
  if trim(coalesce(p_store_id,'')) in ('', 'main_store') then raise exception 'Workspace utama dilindungi.'; end if;
  delete from public.anugerah_store_history where store_id = trim(p_store_id);
  delete from public.anugerah_store where id = trim(p_store_id);
  return jsonb_build_object('status', 'deleted');
end; $$;
grant execute on function public.delete_store_workspace(text, text) to anon, authenticated;

create or replace function public.get_store_snapshots(p_store_id text default 'main_store')
returns table (history_id bigint, store_id text, label text, created_at timestamptz)
language sql security definer set search_path = public as $$
  select h.history_id, h.store_id, h.label, h.created_at from public.anugerah_store_history h
  where h.store_id = coalesce(nullif(trim(p_store_id), ''), 'main_store') order by h.created_at desc limit 20;
$$;
grant execute on function public.get_store_snapshots(text) to anon, authenticated;

create or replace function public.restore_store_snapshot(p_pin_hash text, p_history_id bigint)
returns jsonb language plpgsql security definer set search_path = public as $$
declare v_h text; v_s record;
begin
  select pin_hash into v_h from public.anugerah_admin_auth where id = 'default';
  if v_h is not null and v_h != p_pin_hash then raise exception 'PIN Tidak Valid!'; end if;
  select store_id, label, payload into v_s from public.anugerah_store_history where history_id = p_history_id;
  if not found then raise exception 'Snapshot tidak ditemukan.'; end if;
  insert into public.anugerah_store (id, payload, pin_hash, updated_at) values (v_s.store_id, v_s.payload, null, now())
  on conflict (id) do update set payload = excluded.payload, updated_at = now();
  insert into public.anugerah_store_history (store_id, label, payload)
  values (v_s.store_id, 'Rollback ke #' || p_history_id::text, v_s.payload);
  return jsonb_build_object('status', 'restored', 'store_id', v_s.store_id, 'payload', v_s.payload, 'updated_at', now());
end; $$;
grant execute on function public.restore_store_snapshot(text, bigint) to anon, authenticated;

create or replace function public.log_store_event(p_store_id text default 'main_store', p_event_type text default 'page_view', p_label text default '')
returns void language plpgsql security definer set search_path = public as $$
begin
  insert into public.anugerah_analytics (store_id, event_type, label)
  values (coalesce(nullif(trim(p_store_id),''),'main_store'), left(coalesce(nullif(trim(p_event_type),''),'page_view'),40), left(coalesce(p_label,''),120));
  delete from public.anugerah_analytics where created_at < now() - interval '90 days';
end; $$;
grant execute on function public.log_store_event(text, text, text) to anon, authenticated;

create or replace function public.get_store_analytics(p_pin_hash text, p_store_id text default 'main_store')
returns jsonb language plpgsql security definer set search_path = public as $$
declare v_h text; v_id text := coalesce(nullif(trim(p_store_id),''),'main_store'); v_tot bigint; v_day bigint; v_wa bigint;
begin
  select pin_hash into v_h from public.anugerah_admin_auth where id = 'default';
  if v_h is not null and v_h != p_pin_hash then raise exception 'PIN Tidak Valid!'; end if;
  select count(*) into v_tot from public.anugerah_analytics where store_id = v_id and event_type = 'page_view';
  select count(*) into v_day from public.anugerah_analytics where store_id = v_id and event_type = 'page_view' and created_at >= date_trunc('day', now());
  select count(*) into v_wa from public.anugerah_analytics where store_id = v_id and event_type in ('wa_chat','wa_checkout');
  return jsonb_build_object('views_total', coalesce(v_tot,0), 'views_today', coalesce(v_day,0), 'wa_clicks', coalesce(v_wa,0));
end; $$;
grant execute on function public.get_store_analytics(text, text) to anon, authenticated;

create or replace function public.get_backend_health()
returns jsonb language plpgsql security definer set search_path = public, storage as $$
declare v_rt boolean := false; v_bk boolean := false; v_ws bigint := 0; v_sn bigint := 0;
begin
  select exists (select 1 from pg_publication_tables where pubname = 'supabase_realtime' and schemaname = 'public' and tablename = 'anugerah_store') into v_rt;
  select exists (select 1 from storage.buckets where id = 'anugerah_media' and public = true) into v_bk;
  select count(*) into v_ws from public.anugerah_store;
  select count(*) into v_sn from public.anugerah_store_history;
  return jsonb_build_object('status', 'ok', 'version', '2.2-pro', 'realtime_enabled', v_rt, 'storage_bucket_ready', v_bk, 'workspace_count', v_ws, 'snapshot_count', v_sn, 'server_time', now());
end; $$;
grant execute on function public.get_backend_health() to anon, authenticated;

do $$
begin
  if exists (select 1 from pg_publication where pubname = 'supabase_realtime')
     and not exists (select 1 from pg_publication_tables where pubname = 'supabase_realtime' and schemaname = 'public' and tablename = 'anugerah_store') then
    alter publication supabase_realtime add table public.anugerah_store;
  end if;
end $$;
